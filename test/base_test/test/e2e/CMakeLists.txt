# Calculator and HAP sample tests use the common fastrpc_test IDL and skeleton.
# Their test cases are ordinary base-test suites; no separate QAIC or DSP build
# path is required.

# Each suite owns an all_tests.c entry point. Discover those manifests so adding
# a suite does not require updating a second central registration list.
file(GLOB _e2e_suite_manifests CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*/all_tests.c"
)
list(SORT _e2e_suite_manifests)

set(_e2e_suites)
foreach(_manifest IN LISTS _e2e_suite_manifests)
    get_filename_component(_suite_dir "${_manifest}" DIRECTORY)
    get_filename_component(_suite "${_suite_dir}" NAME)
    if(NOT _suite MATCHES "^[a-z0-9_]+$")
        message(FATAL_ERROR
            "E2E suite directory '${_suite}' is not a valid suite name; "
            "use lowercase letters, digits, and underscores")
    endif()
    list(APPEND _e2e_suites "${_suite}")
endforeach()

set(_e2e_suite_dispatch "// Generated by test/base_test/test/e2e/CMakeLists.txt.\n")
string(APPEND _e2e_suite_dispatch
    "// Do not edit; add an all_tests.c file under test/e2e/<suite> instead.\n\n")
foreach(_suite IN LISTS _e2e_suites)
    string(APPEND _e2e_suite_dispatch
        "void run_${_suite}_e2e_tests(void);\n")
endforeach()
string(APPEND _e2e_suite_dispatch "\nvoid run_e2e_suite_tests(void)\n{\n")
foreach(_suite IN LISTS _e2e_suites)
    string(APPEND _e2e_suite_dispatch "    run_${_suite}_e2e_tests();\n")
endforeach()
string(APPEND _e2e_suite_dispatch "}\n")

set(_e2e_suite_dispatch_source "${CMAKE_CURRENT_BINARY_DIR}/e2e_suite_dispatch.c")
file(GENERATE
    OUTPUT "${_e2e_suite_dispatch_source}"
    CONTENT "${_e2e_suite_dispatch}"
)
set(E2E_SUITE_DISPATCH_SOURCE "${_e2e_suite_dispatch_source}" PARENT_SCOPE)

set(_e2e_suite_libs)

add_library(e2e_support STATIC
    common/e2e_session.c
)
target_include_directories(e2e_support PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/common"
    "${CMAKE_SOURCE_DIR}/utils"
    "${CMAKE_SOURCE_DIR}/utils/fastrpc_utils"
    "${FASTRPC_TEST_IDL_INC_DIR}"
    "${FASTRPC_INC_DIR}"
)
target_link_libraries(e2e_support PRIVATE
    common_test_dep
)

foreach(_suite IN LISTS _e2e_suites)
    set(_suite_dir "${CMAKE_CURRENT_SOURCE_DIR}/${_suite}")
    file(GLOB _suite_sources CONFIGURE_DEPENDS "${_suite_dir}/*.c")

    add_library("e2e_${_suite}_all" STATIC ${_suite_sources})
    target_include_directories("e2e_${_suite}_all" PRIVATE
        "${_suite_dir}"
        "${CMAKE_CURRENT_SOURCE_DIR}/common")
    target_link_libraries("e2e_${_suite}_all" PRIVATE
        common_test_dep
        e2e_support)
    list(APPEND _e2e_suite_libs "e2e_${_suite}_all")
endforeach()

set(ALL_SUITE_LIBS ${ALL_SUITE_LIBS} ${_e2e_suite_libs} PARENT_SCOPE)
